######## the test script (run from a copy of the repository, the JMX password and CA in ../hold)

#!/usr/bin/env bash
# The jvm pack's macOS scripts on this Mac: a JVM of its own here without
# JMX (ps and its counters under /var/folders), and JMX to run-jvm.sh's JVMs
# on Fedora by tunnels. Run from ~/r138/vigil; the password and CA in ../hold.
set -u
export JAVA_HOME=$HOME/clis/jdk/Contents/Home
hold=$(cd .. && pwd)/hold
pass=0 fail=0
check() { if [ "$2" = "$3" ]; then pass=$((pass + 1)); printf '  PASS  %s\n' "$1"; else fail=$((fail + 1)); printf '  FAIL  %s: expected [%s] actual [%s]\n' "$1" "$2" "$3"; fi; }
val() { awk -v m="$2" -v j="jvm=$3" '$1 == "metric" && $2 == m { for (i = 4; i <= NF; i++) if ($i == j) v = $3 } END { print v }' "$out/$1"; }
between() { awk -v v="$(val "$1" "$2" "$3")" -v a="$4" -v b="$5" 'BEGIN { print (v ~ /^-?[0-9]+([job.(.*)][0-9]+)?$/ && v + 0 >= a && v + 0 <= b) ? "yes" : "no [" v "]" }'; }

printf 'host     %s, macOS %s\n' "$(hostname -s)" "$(sw_vers -productVersion)"
printf 'java     %s\n' "$("$JAVA_HOME/bin/java" -version 2>&1 | head -n 1)"
out=$(mktemp -d "${TMPDIR:-/tmp}/vigil-jvm-mac.XXXXXX")
"$JAVA_HOME/bin/java" -Dvigil.it=macplain -Xmx128m -XX:+UseSerialGC tests/integration/backends/workloads/jvm/Trouble.java busy leak threadleak > "$out/macplain.log" 2>&1 &
mp=$!
trap 'kill "$mp" 2> /dev/null; rm -rf "$out"' EXIT
sleep 45

export JVM_TARGETS='mac: main=-Dvigil.it=macplain; j8: jmx=localhost:19181; j11: jmxs=localhost:19182; j17: jmxs=localhost:19183; j21: jmxs=localhost:19184; j25: jmxs=localhost:19185'
export JVM_JMX_USER=operator JVM_CACERT=$hold/ca.crt JVM_MBEANS='java.lang:type=Threading/ThreadCount; java.lang:type=Memory/HeapMemoryUsage.used'
JVM_JMX_PASSWORD=$(cat "$hold/pw")
export JVM_JMX_PASSWORD
printf 'jvms     %s\n' "$JVM_TARGETS"

printf '\n# Every job.\n'
for j in $(sed -n 's/^\[job\.\(.*\)\]$/\1/p' packs/jvm/pack.config); do
    printf '\n$ bash packs/jvm/%s.sh\n' "$j"
    bash "packs/jvm/macos/$j.sh" > "$out/$j" 2>&1
    rc=$?
    head -n 120 "$out/$j"
    check "jvm_$j runs" 0 "$rc"
done

printf '\n# This Mac'"'"'s JVM: ps and its counters, no JMX.\n'
check 'found here' 1 "$(val process running mac)"
check 'its memory' yes "$(between process rss_bytes mac 1e6 1e11)"
check 'its CPU time' yes "$(between process cpu_seconds mac 0.1 1e9)"
check 'from the process' 1 "$(val status tier_os mac)"
check 'and its counters' 1 "$(val status tier_counters mac)"
check 'and no JMX' 0 "$(val status tier_jmx mac)"
check 'the heap, from the counters' yes "$(between heap used_bytes mac 1 1e11)"
check 'leaked threads' yes "$(between threads live mac 30 1e6)"
check 'the JIT' yes "$(between jit compile_seconds mac 0.001 1e6)"
check 'safepoints, a share of its time' yes "$(between safepoints stopped_percent_lifetime mac 0 100)"
check 'GC, a share of its time' yes "$(between gc time_percent_lifetime mac 0 100)"
printf '\n# JMX on Fedora, by tunnels.\n'
check 'JMX without TLS' 1 "$(val status tier_jmx j8)"
check 'JMX with TLS' 1 "$(val status tier_jmx j21)"
check 'a deadlock, named' 2 "$(val deadlocks deadlocked j21)"
check 'threads blocked on a lock' yes "$(between contention blocked j17 1 100)"
check 'the busy thread' yes "$(awk '$1 == "metric" && $2 == "cpu_percent" && / jvm=j25 / && / thread=busy-loop/ { print "yes"; exit }' "$out/hot_threads")"
check 'leaking classloaders' yes "$(between classloaders loaders j11 200 1e6)"
printf '\n# Failures.\n'
JVM_JMX_PASSWORD=not-the-password bash packs/jvm/macos/heap.sh > "$out/wrong" 2>&1
check 'a wrong password fails the run' 1 "$?"
check 'and says why' yes "$(grep -q 'refused the login' "$out/wrong" && echo yes || echo no)"
JVM_CACERT='' bash packs/jvm/macos/heap.sh > "$out/untrusted" 2>&1
check 'a certificate it cannot trust fails the run' 1 "$?"
check 'and says why' yes "$(grep -q 'certificate is not trusted' "$out/untrusted" && echo yes || echo no)"
JVM_TARGETS='gone: jmx=127.0.0.1:1; absent: main=-Dvigil.it=nowhere' bash packs/jvm/macos/status.sh > "$out/gone" 2>&1
check 'JMX that does not answer still runs' 0 "$?"
check 'and reads as jmx_up 0' yes "$(grep -q '^metric jmx_up 0 jvm=gone' "$out/gone" && echo yes || echo no)"
printf '%s passed, %s failed\n' "$pass" "$fail"
